cmake_minimum_required(VERSION 3.16)
project(netkeeper_screenshots LANGUAGES C CXX)

get_filename_component(REPO_ROOT "../.." ABSOLUTE)
if(NOT LVGL_PATH)
    set(LVGL_PATH "${REPO_ROOT}/managed_components/lvgl__lvgl")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_PATH}" lvgl)

add_executable(render render.c "${REPO_ROOT}/main/ui.c"
    "${REPO_ROOT}/main/ui_common.c" "${REPO_ROOT}/main/ui_font_18.c")
target_include_directories(render PRIVATE stubs "${REPO_ROOT}/main")
target_link_libraries(render PRIVATE lvgl)
if(NOT MSVC)
    target_link_libraries(render PRIVATE m)
endif()
